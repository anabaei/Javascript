        <summary>  EventEmitter</summary>

* To implement event emitter, need to create a class to extend the EventEmitter class in node, define event handler using `on` and then create an event using `emit` 
```javascript
const express = require('express');
const EventEmitter = require('events');

const app = express();

class MyEmitter extends EventEmitter {}

const myEmitter = new MyEmitter();

myEmitter.on('customEvent', (message) => {
  console.log(`Received message: ${message}`);
});

app.get('/', (req, res) => {
  myEmitter.emit('customEvent', 'Hello from the server!');
  res.send('Event emitted from the server.');
});

const PORT = 4001;
app.listen(PORT, () => {
  console.log(`Server is listening on port ${PORT}`);
});
```

const checkCache = (req, res, next) => {
  const { key } = req.params;

  // Check if data exists in cache
  client.get(key, (err, data) => {
    if (err) throw err;

    if (data !== null) {
      res.send(`Data from cache: ${data}`);
    } else {
      next();
    }
  });
};

  const { key } = req.params;
